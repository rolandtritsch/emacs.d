* Configure mastodon

#+BEGIN_SRC emacs-lisp

  (use-package mastodon
    :ensure t
    :config
    (setq mastodon-instance-url "https://mastodon.ie")
    (setq mastodon-active-user "tedn")  
  )

  (use-package mastodon-alt
    :load-path "packages/mastodon-alt/"
    :config
    (mastodon-alt-tl-activate)
  )

#+END_SRC
