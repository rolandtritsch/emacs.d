* Configure support for CoPilot

#+BEGIN_SRC emacs-lisp

  (use-package s)
  (use-package dash)
  (use-package editorconfig)

  (use-package copilot
    :after (s dash editorconfig)
    :load-path "packages/copilot.el/"
    :bind (
           :map prog-mode-map
           ("C-z c m" . copilot-mode)
           ("C-z c n" . copilot-next-completion)
           ("C-z c a" . copilot-accept-completion)
           )
    )

#+END_SRC
